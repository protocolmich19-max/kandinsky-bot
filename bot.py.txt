from flask import Flask, request
import os
import requests
from io import BytesIO
from PIL import Image, ImageDraw, ImageFont
from telegram import Update
from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes

TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")
KANDINSKY_API_KEY = os.getenv("KANDINSKY_API_KEY")
API_URL = "https://api.fusionbrain.ai/v1/text2image"

BANNED_WORDS = ["голый", "секс", "порно", "убить", "взрыв", "наркотики", "путин"]

def generate_image(prompt: str) -> BytesIO:
    headers = {"Authorization": f"Bearer {KANDINSKY_API_KEY}", "Content-Type": "application/json"}
    payload = {"model": "Kandinsky-3.1", "prompt": prompt, "width": 1024, "height": 1024, "style": "default", "num_images": 1}
    response = requests.post(API_URL, headers=headers, json=payload)
    if response.status_code != 200:
        raise Exception(f"API error: {response.text}")
    image_url = response.json()["images"][0]["url"]
    img = Image.open(BytesIO(requests.get(image_url).content))
    draw = ImageDraw.Draw(img)
    try:
        font = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", 60)
    except:
        font = ImageFont.load_default()
    bbox = draw.textbbox((0, 0), prompt, font=font)
    w = bbox[2] - bbox[0]
    h = bbox[3] - bbox[1]
    x = (1024 - w) // 2
    y = 1024 - h - 60
    draw.rectangle([x-20, y-20, x+w+20, y+h+20], fill=(0, 0, 0, 180))
    draw.text((x, y), prompt, font=font, fill="white")
    output = BytesIO()
    img.save(output, format='PNG')
    output.seek(0)
    return output

async def start(update: Update, context):
    await update.message.reply_text("Привет! Напиши промпт на русском — сгенерирую картинку!")

async def handle(update: Update, context):
    text = update.message.text.strip()
    if any(word in text.lower() for word in BANNED_WORDS):
        await update.message.reply_text("❌ Запрещённый запрос.")
        return
    await update.message.reply_chat_action('upload_photo')
    try:
        photo = generate_image(text)
        await update.message.reply_photo(photo=photo, caption=f"🎨 {text}")
    except Exception as e:
        await update.message.reply_text(f"⚠️ Ошибка: {str(e)}")

app = Flask(__name__)
application = Application.builder().token(TELEGRAM_TOKEN).build()
application.add_handler(CommandHandler("start", start))
application.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle))

@app.route('/webhook', methods=['POST'])
def webhook():
    update = Update.de_json(request.get_json(), application.bot)
    application.process_update(update)
    return 'OK', 200

if __name__ == '__main__':
    port = int(os.environ.get('PORT', 8080))
    app.run(host='0.0.0.0', port=port)
